cmake_minimum_required(VERSION 3.16)
project(weatherstar_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_image SDL2_ttf)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(XML REQUIRED IMPORTED_TARGET libxml-2.0)
pkg_check_modules(HTTP REQUIRED IMPORTED_TARGET cpp-httplib)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
add_executable(weather-render src/renderer.cpp)
target_link_libraries(weather-render PRIVATE PkgConfig::SDL)
add_executable(weather-collect src/collector.cpp)
target_link_libraries(weather-collect PRIVATE PkgConfig::CURL PkgConfig::XML)
add_executable(weather-serve src/server.cpp)
target_link_libraries(weather-serve PRIVATE PkgConfig::HTTP Threads::Threads OpenSSL::SSL OpenSSL::Crypto)
